   273→function putNode(draft: IrisDoc, id: ID, patch: Partial<IrisNode> | ((n: IrisNode) => Partial<IrisNode>)): void {
   274→  const old = draft.nodesById[id];
   275→  if (!old) return;
   276→  draft.nodesById[id] = { ...old, ...(typeof patch === 'function' ? patch(old) : patch) };
   277→}

   279→function putPage(draft: IrisDoc, id: ID, patch: Partial<{ name: string; nodeIds: ID[] }>): void {
   280→  const old = draft.pagesById[id];
   281→  if (!old) return;
   282→  draft.pagesById[id] = { ...old, ...patch };
   283→}

   285→function removeChildRef(draft: IrisDoc, pageId: ID, parentId: ID | null, childId: ID): void {
   286→  if (parentId === null) {
   287→    const page = draft.pagesById[pageId];
   288→    if (page) putPage(draft, pageId, { nodeIds: page.nodeIds.filter((c) => c !== childId) });
   289→  } else {
   290→    const p = draft.nodesById[parentId];
   291→    if (p) putNode(draft, parentId, { children: p.children.filter((c) => c !== childId) });
   292→  }
   293→}
   294→
   295→function insertChildRef(draft: IrisDoc, pageId: ID, parentId: ID | null, childId: ID, index?: number): void {
   296→  if (parentId === null) {
   297→    const page = draft.pagesById[pageId];
   298→    if (!page) return;
   299→    const list = page.nodeIds.filter((c) => c !== childId);
   300→    const at = index === undefined || index < 0 || index > list.length ? list.length : index;
   301→    list.splice(at, 0, childId);
   302→    putPage(draft, pageId, { nodeIds: list });
   303→  } else {
   304→    const p = draft.nodesById[parentId];
   305→    if (!p) return;
   306→    const list = p.children.filter((c) => c !== childId);
   307→    const at = index === undefined || index < 0 || index > list.length ? list.length : index;
   308→    list.splice(at, 0, childId);
   309→    putNode(draft, parentId, { children: list });
   310→  }
   311→}

   313→/** Extract rotation/x/y so the node's local matrix equals `m` (node size fixed). */
   314→function applyLocalMatrix(node: IrisNode, m: Mat): void {
   315→  const r = setNodeWorldTransform(node, m);
   316→  node.x = r.x;
   317→  node.y = r.y;
   318→  node.rotation = r.rotation;
   319→}

   321→/** Scale all descendants of a container by (sx, sy) in the container's local space. */
   322→function scaleSubtree(draft: IrisDoc, rootId: ID, sx: number, sy: number, scaleFont: boolean): void {
   323→  const root = draft.nodesById[rootId];
   324→  if (!root || !root.children) return;
   325→  const visit = (id: ID) => {
   326→    const n = draft.nodesById[id];
   327→    if (!n) return;
   328→    const patch: Partial<IrisNode> = {
   329→      x: n.x * sx,
   330→      y: n.y * sy,
   331→      w: Math.max(0.5, n.w * sx),
   332→      h: Math.max(0.5, n.h * sy),
   333→    };
   334→    if (scaleFont) {
   335→      const s = (Math.abs(sx) + Math.abs(sy)) / 2;
   336→      if (Math.abs(s - 1) > 0.001) {
   337→        patch.fontSize = Math.max(1, n.fontSize * s);
   338→        patch.cornerRadius = Math.max(0, n.cornerRadius * s);
   339→      }
   340→    }
   341→    if (n.points.length) {
   342→      patch.points = n.points.map((p) => ({ x: p.x * sx, y: p.y * sy, hx: p.hx * sx, hy: p.hy * sy }));
   343→    }
   344→    putNode(draft, id, patch);
   345→    n.children.forEach(visit);
   346→  };
   347→  root.children.forEach(visit);
   348→}

   357→/** Persist the active document into the library (callers debounce). */
   358→function saveAutosave(state: IrisStore): void {
   359→  if (typeof window === 'undefined') return;
   360→  if (!state.docId) return;
   361→  saveDocToLibrary(state.docId, state.doc, {
   362→    currentPageId: state.currentPageId,
   363→    viewport: state.viewport,
   364→  });

   367→function scheduleAutosave(get: () => IrisStore): void {
   368→  if (autosaveTimer) clearTimeout(autosaveTimer);
   369→  autosaveTimer = setTimeout(() => saveAutosave(get()), 800);

   379→/**
   380→ * One-time boot: run the legacy storage migration, then restore the document
   381→ * named in the URL (?doc=<id>). Returns null when starting at the homepage.
   382→ */
   383→function bootIris(): BootState | null {
   384→  if (typeof window === 'undefined') return null;
   385→  migrateLegacyLibrary();
   386→  const route = routeFromLocation();
   387→  if (route.kind !== 'doc') return null;
   388→  const stored = getStoredDoc(route.id);
   389→  if (!stored) {
   390→    replaceRoute({ kind: 'home' }); // stale URL — clean it up
   391→    return null;
   392→  }
   393→  const doc = stored.doc;
   394→  const pageId =
   395→    stored.currentPageId && doc.pages.includes(stored.currentPageId) ? stored.currentPageId : doc.pages[0];
   396→  return { doc, docId: route.id, pageId, viewport: stored.viewport ?? DEFAULT_VIEWPORT };
   397→}

   401→export const useIris = create<IrisStore>()((set, get) => {
   402→  // NOTE: the initial doc must be created exactly once — a second
   403→  // createDefaultDoc() would produce a phantom currentPageId.
   404→  const booted = bootIris();
   405→  const initialDoc = booted?.doc ?? createDefaultDoc();
   406→  const initialPageId = booted?.pageId ?? initialDoc.pages[0];
   407→  const initialRoute: IrisRoute = booted ? { kind: 'doc', id: booted.docId } : { kind: 'home' };
   408→  /**
   409→   * Core mutation pipeline: optional history push, clone-on-write doc mutation,
   410→   * selection sanity, autosave scheduling.
   411→   */
   412→  const commit = (
   413→    mutator: (draft: IrisDoc) => void,
   414→    opts?: { history?: boolean | string; selection?: ID[]; skipDirty?: boolean },
   415→  ): void => {
   416→    const state = get();
   417→    const history = opts?.history === undefined ? true : opts.history;
   418→    if (history !== false) {
   419→      const label = typeof history === 'string' ? history : undefined;
   420→      const now = Date.now();
   421→      const coalesce = label !== undefined && label === lastHistoryLabel && now - lastHistoryAt < 700;
   422→      if (!coalesce && state.doc.nodesById && state.doc.pagesById) {
   423→        const snap: Snapshot = {
   424→          doc: deepClone(state.doc),
   425→          selection: [...state.selection],
   426→          currentPageId: state.currentPageId,
   427→        };
   428→        set((s) => ({ past: [...s.past, snap].slice(-HISTORY_LIMIT), future: [] }));
   429→      }
   430→      lastHistoryLabel = label;
   431→      lastHistoryAt = now;
   432→    }
   433→    const draft = draftOf(state.doc);
   434→    mutator(draft);
   435→    const pageId = get().currentPageId;
   436→    let sel = get().selection.filter((id) => {
   437→      const n = draft.nodesById[id];
   438→      return !!n && n.pageId === pageId;
   439→    });
   440→    if (opts?.selection) {
   441→      const valid = opts.selection.filter((id) => draft.nodesById[id] && draft.nodesById[id].pageId === pageId);
   442→      if (valid.length || opts.selection.length === 0) sel = valid;
   443→    }
   444→    const st = get();
   445→    const editingTextId = st.editingTextId && draft.nodesById[st.editingTextId] ? st.editingTextId : null;
   446→    const editingPathId = st.editingPathId && draft.nodesById[st.editingPathId] ? st.editingPathId : null;
   447→    set({ doc: draft, selection: sel, editingTextId, editingPathId, dirty: opts?.skipDirty ? st.dirty : true });
   448→    scheduleAutosave(get);
   449→  };
   450→
   451→  const currentPage = (): ID => get().currentPageId;
   452→
   453→  /** Shared paste/duplicate: clone source subtrees with a world-space offset. */
   454→  const pasteOrDuplicate = (sourceNodes: IrisNode[], sourceRoots: IrisNode[], dx: number, dy: number): ID[] => {
   455→    if (sourceRoots.length === 0) return [];
   456→    const state = get();
   457→    const pageId = currentPage();
   458→    const idMap = new Map<ID, ID>();
   459→    let counter = 0;
   460→    for (const n of sourceNodes) {
   461→      counter += 1;
   462→      idMap.set(n.id, `${n.id}c${counter}${Math.random().toString(36).slice(2, 6)}`);
   463→    }
   464→    const cloned = sourceNodes.map((n) => ({
   465→      ...n,
   466→      id: idMap.get(n.id)!,
   467→      pageId,
   468→      children: n.children.map((c) => idMap.get(c)).filter((c): c is ID => !!c),
   469→      points: n.points.map((p) => ({ ...p })),
   470→      fillGradient: n.fillGradient ? { ...n.fillGradient } : null,
   471→      shadow: n.shadow ? { ...n.shadow } : null,
   472→    }));
   473→    const worldTargets = new Map<ID, Mat>();
   474→    for (const root of sourceRoots) {
   475→      const W = nodeWorldMatrix(state.doc, root);
   476→      worldTargets.set(idMap.get(root.id)!, mMul(mTranslate(dx, dy), W));
   477→    }
   478→    const newRootIds = sourceRoots.map((r) => idMap.get(r.id)!);
   479→    commit(
   480→      (draft) => {
   481→        for (const c of cloned) {
   482→          let parentId: ID | null = null;
   483→          if (c.parentId) {
   484→            const pid = idMap.get(c.parentId) ?? c.parentId;
   485→            const pn = draft.nodesById[pid];
   486→            if (pn && pn.pageId === pageId && isContainer(pn)) parentId = pid;
   487→          }
   488→          const node: IrisNode = { ...c, parentId };
   489→          const target = worldTargets.get(c.id);
   490→          if (target) applyLocalMatrix(node, target);
   491→          draft.nodesById[c.id] = node;
   492→        }
   493→        for (const c of cloned) {
   494→          const node = draft.nodesById[c.id];
   495→          if (node) insertChildRef(draft, pageId, node.parentId, c.id, undefined);
   496→        }
   497→      },
   498→      { history: 'paste', selection: newRootIds },
   499→    );
   500→    return newRootIds;
   501→  };
   502→
   503→  const viewportCenter = (): { x: number; y: number } => {
   504→    const { viewport, canvasSize } = get();
   505→    return screenToWorld(viewport, Math.max(0, canvasSize.w / 2 - 10), Math.max(0, canvasSize.h / 2 - 10));
   506→  };
   507→
   508→  /** Style fields copied by copy/paste style. */
   509→  const STYLE_FIELDS: (keyof IrisNode)[] = [
   510→    'fill',
   511→    'stroke',
   512→    'strokeWidth',
   513→    'cornerRadius',
   514→    'opacity',
   515→    'dash',
   516→    'fillGradient',
   517→    'shadow',
   518→    'blur',
   519→    'blend',
   520→    'fontFamily',
   521→    'fontSize',
   522→    'fontWeight',
   523→    'textAlign',
   524→    'lineHeight',
   525→    'letterSpacing',
   526→    'textCase',
   527→    'underline',
   528→    'strike',
   529→  ];
   530→
   531→  return {
   532→    doc: initialDoc,
   533→    docId: booted?.docId ?? null,
   534→    route: initialRoute,
   535→    currentPageId: initialPageId,
   536→    selection: [],
   537→    tool: 'move',
   538→    viewport: booted?.viewport ?? DEFAULT_VIEWPORT,
   539→    editingTextId: null,
   540→    editingPathId: null,
   541→    hoverId: null,
   542→    viewFlags: { rulers: true, pixelGrid: true, outline: false },
   543→    past: [],
   544→    future: [],
   545→    clipboard: null,
   546→    contextMenu: null,
   547→    exportOpen: false,
   548→    shortcutsOpen: false,
   549→    toasts: [],
   550→    savedName: null,
   551→    dirty: false,
   552→    canvasSize: { w: 800, h: 600 },
   553→    styleClipboard: null,
   554→
   555→    /* ------------------------- tool / view ------------------------- */
   556→    setTool: (tool) => {
   557→      const st = get();
   558→      if (st.tool === tool) return;
   559→      if (st.editingTextId) set({ editingTextId: null });
   560→      if (st.editingPathId) set({ editingPathId: null });

  1033→          for (const n of ordered) {

  1150→    alignSelection: (mode) => {
  1151→      const state = get();
  1152→      const nodes = state.selection.map((id) => state.doc.nodesById[id]).filter(Boolean) as IrisNode[];
  1153→      if (nodes.length === 0) return;
  1154→      const doc = state.doc;
  1155→      const rects = new Map<ID, Rect>(nodes.map((n) => [n.id, nodeWorldAABB(doc, n)]));
  1156→      let ref: Rect;

  1158→        ref = unionRects([...rects.values()]);

  1160→        // single object: align within its parent frame (no-op at page top level, like Figma)
  1161→        const n = nodes[0];

  1166→      }

  1169→        if (nodes.length < 2) return;
  1170→        const horiz = mode === 'hdist';
  1171→        const sorted = nodes.slice().sort((a, b) => {
  1172→          const ra = rects.get(a.id)!;
  1173→          const rb = rects.get(b.id)!;
  1174→          return horiz ? ra.x - rb.x : ra.y - rb.y;
  1175→        });
  1176→        const first = rects.get(sorted[0].id)!;
  1177→        const last = rects.get(sorted[sorted.length - 1].id)!;
  1178→        const span = horiz ? last.x + last.w - first.x : last.y + last.h - first.y;
  1179→        const total = sorted.reduce((acc, n) => acc + (horiz ? rects.get(n.id)!.w : rects.get(n.id)!.h), 0);
  1180→        const gap = (span - total) / (sorted.length - 1);
  1181→        let cursor = horiz ? first.x : first.y;
  1182→        for (const n of sorted) {
  1183→          const r = rects.get(n.id)!;
  1184→          const target = cursor;
  1185→          const cur = horiz ? r.x : r.y;
  1186→          moves.push({ id: n.id, dx: horiz ? target - cur : 0, dy: horiz ? 0 : target - cur });
  1187→          cursor += (horiz ? r.w : r.h) + gap;
  1188→        }
  1189→      } else {
  1190→        for (const n of nodes) {
  1191→          const r = rects.get(n.id)!;
  1192→          let dx = 0;
  1193→          let dy = 0;
  1194→          if (mode === 'left') dx = ref.x - r.x;
  1195→          if (mode === 'hcenter') dx = ref.x + ref.w / 2 - (r.x + r.w / 2);
  1196→          if (mode === 'right') dx = ref.x + ref.w - (r.x + r.w);
  1197→          if (mode === 'top') dy = ref.y - r.y;
  1198→          if (mode === 'vcenter') dy = ref.y + ref.h / 2 - (r.y + r.h / 2);
  1199→          if (mode === 'bottom') dy = ref.y + ref.h - (r.y + r.h);
  1200→          moves.push({ id: n.id, dx, dy });
  1201→        }
  1202→      }
  1203→      commit(
  1204→        (draft) => {
  1205→          for (const mv of moves) {
  1206→            const n = draft.nodesById[mv.id];
  1207→            if (!n || (mv.dx === 0 && mv.dy === 0)) continue;
  1208→            const P = parentWorldMatrix(draft, n);
  1209→            const local = mApplyVec(mInvert(P), mv.dx, mv.dy);
  1210→            putNode(draft, mv.id, { x: n.x + local.x, y: n.y + local.y });

  1212→        },

  1538→        docId: id,
  1539→        route: { kind: 'doc', id },
  1540→        currentPageId: doc.pages[0],
  1541→        selection: [],
  1542→        past: [],
  1543→        future: [],
  1544→        clipboard: null,
  1545→        editingTextId: null,
  1546→        editingPathId: null,
  1547→        savedName: null,
  1548→        dirty: false,
  1549→        viewport: DEFAULT_VIEWPORT,
  1550→      });
  1551→      pushRoute({ kind: 'doc', id });
  1552→      get().toast('New document created');
  1553→    },
  1554→
  1555→    newDocFromTemplate: (templateId) => {
  1556→      get().flushSave();
  1557→      const doc = buildTemplate(templateId);
  1558→      const entry = importDocIntoLibrary(doc);
  1559→      set({
  1560→        doc,
  1561→        docId: entry.id,
  1562→        route: { kind: 'doc', id: entry.id },
  1563→        currentPageId: doc.pages[0],
  1564→        selection: [],
  1565→        past: [],
  1566→        future: [],
  1567→        clipboard: null,
  1568→        editingTextId: null,
  1569→        editingPathId: null,
  1570→        savedName: null,
  1571→        dirty: false,
  1572→        viewport: { x: 60, y: 60, zoom: 0.75 },
  1573→      });
  1574→      pushRoute({ kind: 'doc', id: entry.id });
  1575→      get().toast(`Created “${doc.name}”`);
  1576→    },
  1577→
  1578→    /**
  1579→     * Open an external document (.iris file / recent) — registers it as a new
  1580→     * library entry so it shows up on the homepage and can be reopened later.
  1581→     */
  1582→    loadDoc: (rawDoc, savedName, fitView) => {
  1583→      get().flushSave();
  1584→      const doc = normalizeDoc(rawDoc);
  1585→      if (savedName) doc.name = savedName;
  1586→      const entry = importDocIntoLibrary(doc);
  1587→      set({
  1588→        doc,
  1589→        docId: entry.id,
  1590→        route: { kind: 'doc', id: entry.id },
  1591→        currentPageId: doc.pages[0],
  1592→        selection: [],
  1593→        past: [],
  1594→        future: [],
  1595→        clipboard: null,
  1596→        editingTextId: null,
  1597→        editingPathId: null,
  1598→        savedName: savedName ?? null,
  1599→        dirty: false,
  1600→        viewport: DEFAULT_VIEWPORT,
  1601→      });
  1602→      pushRoute({ kind: 'doc', id: entry.id });
  1603→      if (fitView) get().zoomToFit();
  1604→    },
  1605→
  1606→    /** Open a document from the library (homepage card click). */
  1607→    openDoc: (id, opts) => {
  1608→      const s = get();
  1609→      if (s.route.kind === 'doc' && s.route.id === id) return;
  1610→      s.flushSave();
  1611→      const stored = getStoredDoc(id);
  1612→      if (!stored) {
  1613→        get().toast('That file could not be opened');
  1614→        return;
  1615→      }
  1616→      const doc = stored.doc;
  1617→      const pageId =
  1618→        stored.currentPageId && doc.pages.includes(stored.currentPageId) ? stored.currentPageId : doc.pages[0];
  1619→      set({
  1620→        doc,
  1621→        docId: id,
  1622→        route: { kind: 'doc', id },
  1623→        currentPageId: pageId,
  1624→        selection: [],
  1625→        past: [],
  1626→        future: [],
  1627→        clipboard: null,
  1628→        editingTextId: null,
  1629→        editingPathId: null,
  1630→        savedName: null,
  1631→        dirty: false,
  1632→        viewport: stored.viewport ?? DEFAULT_VIEWPORT,
  1633→      });
  1634→      if (opts?.push !== false) pushRoute({ kind: 'doc', id });
  1635→    },
  1636→
  1637→    /** Leave the editor and return to the file browser (homepage). */
  1638→    goHome: (opts) => {
  1639→      get().flushSave();
  1640→      set({
  1641→        route: { kind: 'home' },
  1642→        docId: null,
  1643→        selection: [],
  1644→        editingTextId: null,
  1645→        editingPathId: null,
  1646→        contextMenu: null,
  1647→      });
  1648→      if (opts?.push !== false) pushRoute({ kind: 'home' });
  1649→    },
  1650→
  1651→    /** Browser Back/Forward: adopt the URL route without pushing history. */
  1652→    syncRoute: () => {
  1653→      const r = routeFromLocation();
  1654→      const s = get();
  1655→      if (r.kind === 'doc') {
  1656→        if (s.route.kind === 'doc' && s.route.id === r.id) return;
  1657→        if (!getStoredDoc(r.id)) {
  1658→          replaceRoute({ kind: 'home' });
  1659→          if (s.route.kind !== 'home') get().goHome({ push: false });
  1660→          get().toast('That file is no longer available');
  1661→          return;
  1662→        }
  1663→        get().openDoc(r.id, { push: false });
  1664→        return;
  1665→      }
  1666→      if (s.route.kind === 'home') return;
  1667→      get().goHome({ push: false });
  1668→    },
  1669→
  1670→    markSaved: (name) => set({ savedName: name, dirty: false }),
  1671→
  1672→    /* ------------------------------ ui ----------------------------- */
  1673→    toast: (msg, action) => {
  1674→      const id = toastSeq++;
  1675→      set((s) => ({ toasts: [...s.toasts.slice(-3), { id, msg, action }] }));
  1676→      setTimeout(() => get().dismissToast(id), action ? 5000 : 2400);
  1677→    },
  1678→    dismissToast: (id) => set((s) => ({ toasts: s.toasts.filter((t) => t.id !== id) })),
  1679→    openContextMenu: (x, y, worldX, worldY) => set({ contextMenu: { x, y, worldX, worldY } }),
  1680→    closeContextMenu: () => set({ contextMenu: null }),
  1681→    setExportOpen: (open) => set({ exportOpen: open }),
  1682→    setShortcutsOpen: (open) => set({ shortcutsOpen: open }),
  1683→  };
  1684→});
