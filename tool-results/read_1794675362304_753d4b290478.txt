   320→function putNode(draft: IrisDoc, id: ID, patch: Partial<IrisNode> | ((n: IrisNode) => Partial<IrisNode>)): void {
   321→  const old = draft.nodesById[id];
   322→  if (!old) return;
   323→  draft.nodesById[id] = { ...old, ...(typeof patch === 'function' ? patch(old) : patch) };
   324→}

   326→function putPage(draft: IrisDoc, id: ID, patch: Partial<{ name: string; nodeIds: ID[] }>): void {
   327→  const old = draft.pagesById[id];
   328→  if (!old) return;
   329→  draft.pagesById[id] = { ...old, ...patch };

   332→function removeChildRef(draft: IrisDoc, pageId: ID, parentId: ID | null, childId: ID): void {
   333→  if (parentId === null) {
   334→    const page = draft.pagesById[pageId];
   335→    if (page) putPage(draft, pageId, { nodeIds: page.nodeIds.filter((c) => c !== childId) });
   336→  } else {
   337→    const p = draft.nodesById[parentId];
   338→    if (p) putNode(draft, parentId, { children: p.children.filter((c) => c !== childId) });
   339→  }
   340→}
   341→
   342→function insertChildRef(draft: IrisDoc, pageId: ID, parentId: ID | null, childId: ID, index?: number): void {
   343→  if (parentId === null) {
   344→    const page = draft.pagesById[pageId];
   345→    if (!page) return;
   346→    const list = page.nodeIds.filter((c) => c !== childId);
   347→    const at = index === undefined || index < 0 || index > list.length ? list.length : index;
   348→    list.splice(at, 0, childId);
   349→    putPage(draft, pageId, { nodeIds: list });
   350→  } else {
   351→    const p = draft.nodesById[parentId];
   352→    if (!p) return;
   353→    const list = p.children.filter((c) => c !== childId);
   354→    const at = index === undefined || index < 0 || index > list.length ? list.length : index;
   355→    list.splice(at, 0, childId);
   356→    putNode(draft, parentId, { children: list });
   357→  }
   358→}

   360→/** Extract rotation/x/y so the node's local matrix equals `m` (node size fixed). */
   361→function applyLocalMatrix(node: IrisNode, m: Mat): void {
   362→  const r = setNodeWorldTransform(node, m);
   363→  node.x = r.x;
   364→  node.y = r.y;
   365→  node.rotation = r.rotation;

   368→/** Scale all descendants of a container by (sx, sy) in the container's local space. */
   369→function scaleSubtree(draft: IrisDoc, rootId: ID, sx: number, sy: number, scaleFont: boolean): void {
   370→  const root = draft.nodesById[rootId];
   371→  if (!root || !root.children) return;
   372→  const visit = (id: ID) => {
   373→    const n = draft.nodesById[id];
   374→    if (!n) return;
   375→    const patch: Partial<IrisNode> = {
   376→      x: n.x * sx,
   377→      y: n.y * sy,
   378→      w: Math.max(0.5, n.w * sx),
   379→      h: Math.max(0.5, n.h * sy),
   380→    };
   381→    if (scaleFont) {
   382→      const s = (Math.abs(sx) + Math.abs(sy)) / 2;
   383→      if (Math.abs(s - 1) > 0.001) {
   384→        patch.fontSize = Math.max(1, n.fontSize * s);
   385→        patch.cornerRadius = Math.max(0, n.cornerRadius * s);
   386→      }
   387→    }
   388→    if (n.points.length) {
   389→      patch.points = n.points.map((p) => ({ x: p.x * sx, y: p.y * sy, hx: p.hx * sx, hy: p.hy * sy }));
   390→    }
   391→    putNode(draft, id, patch);
   392→    n.children.forEach(visit);
   393→  };
   394→  root.children.forEach(visit);
   395→}

   404→/** Persist the active document into the library (callers debounce). */
   405→function saveAutosave(state: IrisStore): void {
   406→  if (typeof window === 'undefined') return;
   407→  if (!state.docId) return;
   408→  saveDocToLibrary(state.docId, state.doc, {
   409→    currentPageId: state.currentPageId,
   410→    viewport: state.viewport,
   411→  });
   412→}

   414→function scheduleAutosave(get: () => IrisStore): void {
   415→  if (autosaveTimer) clearTimeout(autosaveTimer);
   416→  autosaveTimer = setTimeout(() => saveAutosave(get()), 800);
   417→}

   426→/**
   427→ * One-time boot: run the legacy storage migration, then restore the document
   428→ * named in the URL (?doc=<id>). Returns null when starting at the homepage.
   429→ */
   430→function bootIris(): BootState | null {
   431→  if (typeof window === 'undefined') return null;
   432→  migrateLegacyLibrary();
   433→  const route = routeFromLocation();
   434→  if (route.kind !== 'doc') return null;
   435→  const stored = getStoredDoc(route.id);
   436→  if (!stored) {
   437→    replaceRoute({ kind: 'home' }); // stale URL — clean it up
   438→    return null;
   439→  }
   440→  const doc = stored.doc;
   441→  const pageId =
   442→    stored.currentPageId && doc.pages.includes(stored.currentPageId) ? stored.currentPageId : doc.pages[0];
   443→  return { doc, docId: route.id, pageId, viewport: stored.viewport ?? DEFAULT_VIEWPORT };
   444→}

   448→export const useIris = create<IrisStore>()((set, get) => {
   449→  // NOTE: the initial doc must be created exactly once — a second
   450→  // createDefaultDoc() would produce a phantom currentPageId.
   451→  const booted = bootIris();
   452→  const initialDoc = booted?.doc ?? createDefaultDoc();
   453→  const initialPageId = booted?.pageId ?? initialDoc.pages[0];
   454→  const initialRoute: IrisRoute = booted ? { kind: 'doc', id: booted.docId } : { kind: 'home' };
   455→  /**
   456→   * Core mutation pipeline: optional history push, clone-on-write doc mutation,
   457→   * selection sanity, autosave scheduling.
   458→   */
   459→  const commit = (
   460→    mutator: (draft: IrisDoc) => void,
   461→    opts?: { history?: boolean | string; selection?: ID[]; skipDirty?: boolean },
   462→  ): void => {
   463→    const state = get();
   464→    const history = opts?.history === undefined ? true : opts.history;
   465→    if (history !== false) {
   466→      const label = typeof history === 'string' ? history : undefined;
   467→      const now = Date.now();
   468→      const coalesce = label !== undefined && label === lastHistoryLabel && now - lastHistoryAt < 700;
   469→      if (!coalesce && state.doc.nodesById && state.doc.pagesById) {
   470→        const snap: Snapshot = {
   471→          doc: deepClone(state.doc),
   472→          selection: [...state.selection],
   473→          currentPageId: state.currentPageId,
   474→        };
   475→        set((s) => ({ past: [...s.past, snap].slice(-HISTORY_LIMIT), future: [] }));
   476→      }
   477→      lastHistoryLabel = label;
   478→      lastHistoryAt = now;
   479→    }
   480→    const draft = draftOf(state.doc);
   481→    mutator(draft);
   482→    const pageId = get().currentPageId;
   483→    let sel = get().selection.filter((id) => {
   484→      const n = draft.nodesById[id];
   485→      return !!n && n.pageId === pageId;
   486→    });
   487→    if (opts?.selection) {
   488→      const valid = opts.selection.filter((id) => draft.nodesById[id] && draft.nodesById[id].pageId === pageId);
   489→      if (valid.length || opts.selection.length === 0) sel = valid;
   490→    }
   491→    const st = get();
   492→    const editingTextId = st.editingTextId && draft.nodesById[st.editingTextId] ? st.editingTextId : null;
   493→    const editingPathId = st.editingPathId && draft.nodesById[st.editingPathId] ? st.editingPathId : null;
   494→    set({ doc: draft, selection: sel, editingTextId, editingPathId, dirty: opts?.skipDirty ? st.dirty : true });
   495→    scheduleAutosave(get);
   496→  };
   497→
   498→  const currentPage = (): ID => get().currentPageId;
   499→
   500→  /** Shared paste/duplicate: clone source subtrees with a world-space offset. */
   501→  const pasteOrDuplicate = (sourceNodes: IrisNode[], sourceRoots: IrisNode[], dx: number, dy: number): ID[] => {
   502→    if (sourceRoots.length === 0) return [];
   503→    const state = get();
   504→    const pageId = currentPage();
   505→    const idMap = new Map<ID, ID>();
   506→    let counter = 0;
   507→    for (const n of sourceNodes) {
   508→      counter += 1;
   509→      idMap.set(n.id, `${n.id}c${counter}${Math.random().toString(36).slice(2, 6)}`);
   510→    }
   511→    const cloned = sourceNodes.map((n) => ({
   512→      ...n,
   513→      id: idMap.get(n.id)!,
   514→      pageId,
   515→      children: n.children.map((c) => idMap.get(c)).filter((c): c is ID => !!c),
   516→      points: n.points.map((p) => ({ ...p })),
   517→      fillGradient: n.fillGradient ? { ...n.fillGradient } : null,
   518→      shadow: n.shadow ? { ...n.shadow } : null,
   519→    }));
   520→    const worldTargets = new Map<ID, Mat>();
   521→    for (const root of sourceRoots) {
   522→      const W = nodeWorldMatrix(state.doc, root);
   523→      worldTargets.set(idMap.get(root.id)!, mMul(mTranslate(dx, dy), W));
   524→    }
   525→    const newRootIds = sourceRoots.map((r) => idMap.get(r.id)!);
   526→    commit(
   527→      (draft) => {
   528→        for (const c of cloned) {
   529→          let parentId: ID | null = null;
   530→          if (c.parentId) {
   531→            const pid = idMap.get(c.parentId) ?? c.parentId;
   532→            const pn = draft.nodesById[pid];
   533→            if (pn && pn.pageId === pageId && isContainer(pn)) parentId = pid;
   534→          }
   535→          const node: IrisNode = { ...c, parentId };
   536→          const target = worldTargets.get(c.id);
   537→          if (target) applyLocalMatrix(node, target);
   538→          draft.nodesById[c.id] = node;
   539→        }
   540→        for (const c of cloned) {
   541→          const node = draft.nodesById[c.id];
   542→          if (node) insertChildRef(draft, pageId, node.parentId, c.id, undefined);
   543→        }
   544→      },
   545→      { history: 'paste', selection: newRootIds },
   546→    );
   547→    return newRootIds;
   548→  };
   549→
   550→  const viewportCenter = (): { x: number; y: number } => {
   551→    const { viewport, canvasSize } = get();
   552→    return screenToWorld(viewport, Math.max(0, canvasSize.w / 2 - 10), Math.max(0, canvasSize.h / 2 - 10));

  1283→    alignSelection: (mode) => {
  1284→      const state = get();
  1285→      const nodes = state.selection.map((id) => state.doc.nodesById[id]).filter(Boolean) as IrisNode[];
  1286→      if (nodes.length === 0) return;
  1287→      const doc = state.doc;
  1288→      const rects = new Map<ID, Rect>(nodes.map((n) => [n.id, nodeWorldAABB(doc, n)]));
  1289→      let ref: Rect;
  1290→      if (nodes.length > 1) {
  1291→        ref = unionRects([...rects.values()]);
  1292→      } else {
  1293→        // single object: align within its parent frame (no-op at page top level, like Figma)
  1294→        const n = nodes[0];
  1295→        let p = n.parentId ? doc.nodesById[n.parentId] : undefined;
  1296→        while (p && p.type !== 'frame' && p.parentId) p = doc.nodesById[p.parentId];
  1297→        if (!p || p.type !== 'frame') return;
  1298→        ref = nodeWorldAABB(doc, p);
  1299→      }
  1300→      const moves: { id: ID; dx: number; dy: number }[] = [];

  1302→        if (nodes.length < 2) return;
  1303→        const horiz = mode === 'hdist';
  1304→        const sorted = nodes.slice().sort((a, b) => {
  1305→          const ra = rects.get(a.id)!;
  1306→          const rb = rects.get(b.id)!;
  1307→          return horiz ? ra.x - rb.x : ra.y - rb.y;
  1308→        });
  1309→        const first = rects.get(sorted[0].id)!;
  1310→        const last = rects.get(sorted[sorted.length - 1].id)!;
  1311→        const span = horiz ? last.x + last.w - first.x : last.y + last.h - first.y;
  1312→        const total = sorted.reduce((acc, n) => acc + (horiz ? rects.get(n.id)!.w : rects.get(n.id)!.h), 0);
  1313→        const gap = (span - total) / (sorted.length - 1);
  1314→        let cursor = horiz ? first.x : first.y;
  1315→        for (const n of sorted) {
  1316→          const r = rects.get(n.id)!;
  1317→          const target = cursor;
  1318→          const cur = horiz ? r.x : r.y;
  1319→          moves.push({ id: n.id, dx: horiz ? target - cur : 0, dy: horiz ? 0 : target - cur });
  1320→          cursor += (horiz ? r.w : r.h) + gap;
  1321→        }
  1322→      } else {
  1323→        for (const n of nodes) {
  1324→          const r = rects.get(n.id)!;
  1325→          let dx = 0;
  1326→          let dy = 0;
  1327→          if (mode === 'left') dx = ref.x - r.x;
  1328→          if (mode === 'hcenter') dx = ref.x + ref.w / 2 - (r.x + r.w / 2);
  1329→          if (mode === 'right') dx = ref.x + ref.w - (r.x + r.w);
  1330→          if (mode === 'top') dy = ref.y - r.y;
  1331→          if (mode === 'vcenter') dy = ref.y + ref.h / 2 - (r.y + r.h / 2);
  1332→          if (mode === 'bottom') dy = ref.y + ref.h - (r.y + r.h);
  1333→          moves.push({ id: n.id, dx, dy });
  1334→        }
  1335→      }
  1336→      commit(
  1337→        (draft) => {
  1338→          for (const mv of moves) {
  1339→            const n = draft.nodesById[mv.id];
  1340→            if (!n || (mv.dx === 0 && mv.dy === 0)) continue;
  1341→            const P = parentWorldMatrix(draft, n);
  1342→            const local = mApplyVec(mInvert(P), mv.dx, mv.dy);

  1344→          }
  1345→        },
  1346→        { history: 'align' },
  1347→      );

  1538→        { history: 'pasteStyle' },
  1539→      );
  1540→      get().toast(`Pasted style onto ${state.selection.length} object${state.selection.length === 1 ? '' : 's'}`);
  1541→    },
  1542→
  1543→    /* ------------------------- doc settings ------------------------ */
  1544→    setSettings: (patch) => {

  1550→            grid: { ...draft.settings.grid, ...(patch.grid ?? {}) },
  1551→            guidesX: patch.guidesX ?? [...draft.settings.guidesX],
  1552→            guidesY: patch.guidesY ?? [...draft.settings.guidesY],

  1554→        },

  1562→        },
  1563→        { history: 'settings' },
  1564→      );
  1565→    },
  1566→    addGuide: (axis, at) => {
  1567→      commit(
  1568→        (draft) => {
  1569→          if (axis === 'x') draft.settings.guidesX = [...draft.settings.guidesX, Math.round(at)];
  1570→          else draft.settings.guidesY = [...draft.settings.guidesY, Math.round(at)];
  1571→        },
  1572→        { history: 'guide' },
  1573→      );
  1574→    },
  1575→    moveGuide: (axis, index, at) => {
  1576→      commit(
  1577→        (draft) => {
  1578→          if (axis === 'x') {
  1579→            const list = [...draft.settings.guidesX];
  1580→            if (index < 0 || index >= list.length) return;
  1581→            list[index] = Math.round(at);
  1582→            draft.settings.guidesX = list;
  1583→          } else {
  1584→            const list = [...draft.settings.guidesY];
  1585→            if (index < 0 || index >= list.length) return;
  1586→            list[index] = Math.round(at);
  1587→            draft.settings.guidesY = list;
  1588→          }
  1589→        },
  1590→        { history: false },
  1591→      );
  1592→    },
  1593→    removeGuide: (axis, index) => {
  1594→      commit(
  1595→        (draft) => {
  1596→          if (axis === 'x') draft.settings.guidesX = draft.settings.guidesX.filter((_, i) => i !== index);
  1597→          else draft.settings.guidesY = draft.settings.guidesY.filter((_, i) => i !== index);
  1598→        },
  1599→        { history: 'guide' },
  1600→      );
  1601→    },
  1602→    clearGuides: () => {
  1603→      const s = get();
  1604→      if (!s.doc.settings.guidesX.length && !s.doc.settings.guidesY.length) return;
  1605→      commit(
  1606→        (draft) => {
  1607→          draft.settings.guidesX = [];
  1608→          draft.settings.guidesY = [];
  1609→        },
  1610→        { history: 'guide' },
  1611→      );
  1612→      get().toast('Cleared all guides');
  1613→    },
  1614→
  1615→    /* --------------------------- editing --------------------------- */
  1616→    setEditingText: (id) => {
  1617→      if (id) get().setSelection([id]);
  1618→      set({ editingTextId: id, editingPathId: null });
  1619→    },
  1620→    setEditingPath: (id) => {
  1621→      if (id) get().setSelection([id]);
  1622→      set({ editingPathId: id, editingTextId: null });
  1623→    },
  1624→
  1625→    /* --------------------------- images ---------------------------- */
  1626→    importImage: (src, w, h, at, parentId) => {
  1627→      const state = get();
  1628→      const scale = Math.min(1, 800 / Math.max(w, h, 1));
  1629→      const nw = Math.max(1, Math.round(w * scale));
  1630→      const nh = Math.max(1, Math.round(h * scale));
  1631→      const center = at ?? viewportCenter();
  1632→      let x = center.x - nw / 2;
  1633→      let y = center.y - nh / 2;
  1634→      // when parenting into a frame, convert to parent-local coords
  1635→      if (parentId) {
  1636→        const parent = state.doc.nodesById[parentId];
  1637→        if (parent) {
  1638→          const inv = mInvert(nodeWorldMatrix(state.doc, parent));
  1639→          const local = mApply(inv, x, y);
  1640→          x = local.x;
  1641→          y = local.y;
  1642→        }
  1643→      }
  1644→      return get().addNode(
  1645→        'image',
  1646→        {
  1647→          src,
  1648→          w: nw,
  1649→          h: nh,
  1650→          x,
  1651→          y,
  1652→        },
  1653→        parentId ?? null,
  1654→      );
  1655→    },
  1656→
  1657→    /* ---------------------------- files ---------------------------- */
  1658→    flushSave: () => {
  1659→      if (autosaveTimer) {
  1660→        clearTimeout(autosaveTimer);
  1661→        autosaveTimer = null;
  1662→      }
  1663→      saveAutosave(get());
  1664→    },
  1665→
  1666→    newDoc: () => {
  1667→      get().flushSave();
  1668→      const { id, doc } = createLibraryDoc();
  1669→      set({
  1670→        doc,
  1671→        docId: id,
  1672→        route: { kind: 'doc', id },
  1673→        currentPageId: doc.pages[0],
  1674→        selection: [],
  1675→        past: [],
  1676→        future: [],
  1677→        clipboard: null,
  1678→        editingTextId: null,
  1679→        editingPathId: null,
  1680→        savedName: null,
  1681→        dirty: false,
  1682→        viewport: DEFAULT_VIEWPORT,
  1683→      });
  1684→      pushRoute({ kind: 'doc', id });
  1685→      get().toast('New document created');
  1686→    },
  1687→
  1688→    newDocFromTemplate: (templateId) => {
  1689→      get().flushSave();
  1690→      const doc = buildTemplate(templateId);
  1691→      const entry = importDocIntoLibrary(doc);
  1692→      set({
  1693→        doc,
  1694→        docId: entry.id,
  1695→        route: { kind: 'doc', id: entry.id },
  1696→        currentPageId: doc.pages[0],
  1697→        selection: [],
  1698→        past: [],
  1699→        future: [],
  1700→        clipboard: null,
  1701→        editingTextId: null,
  1702→        editingPathId: null,
  1703→        savedName: null,
  1704→        dirty: false,
  1705→        viewport: { x: 60, y: 60, zoom: 0.75 },
  1706→      });
  1707→      pushRoute({ kind: 'doc', id: entry.id });
  1708→      get().toast(`Created “${doc.name}”`);
  1709→    },
  1710→
  1711→    /**
  1712→     * Open an external document (.iris file / recent) — registers it as a new
  1713→     * library entry so it shows up on the homepage and can be reopened later.
  1714→     */
  1715→    loadDoc: (rawDoc, savedName, fitView) => {
  1716→      get().flushSave();
  1717→      const doc = normalizeDoc(rawDoc);
  1718→      if (savedName) doc.name = savedName;
  1719→      const entry = importDocIntoLibrary(doc);
  1720→      set({
  1721→        doc,
  1722→        docId: entry.id,
  1723→        route: { kind: 'doc', id: entry.id },
  1724→        currentPageId: doc.pages[0],
  1725→        selection: [],
  1726→        past: [],
  1727→        future: [],
  1728→        clipboard: null,
  1729→        editingTextId: null,
  1730→        editingPathId: null,
  1731→        savedName: savedName ?? null,
  1732→        dirty: false,
  1733→        viewport: DEFAULT_VIEWPORT,
  1734→      });
  1735→      pushRoute({ kind: 'doc', id: entry.id });
  1736→      if (fitView) get().zoomToFit();
  1737→    },
  1738→
  1739→    /** Open a document from the library (homepage card click). */
  1740→    openDoc: (id, opts) => {
  1741→      const s = get();
  1742→      if (s.route.kind === 'doc' && s.route.id === id) return;
  1743→      s.flushSave();
  1744→      const stored = getStoredDoc(id);

  1751→        stored.currentPageId && doc.pages.includes(stored.currentPageId) ? stored.currentPageId : doc.pages[0];

  1759→        future: [],

  1785→    syncRoute: () => {
  1786→      const r = routeFromLocation();

  1788→      if (r.kind === 'doc') {
  1789→        if (s.route.kind === 'doc' && s.route.id === r.id) return;

  1791→          replaceRoute({ kind: 'home' });

  1795→        }
  1796→        get().openDoc(r.id, { push: false });
  1797→        return;
  1798→      }
  1799→      if (s.route.kind === 'home') return;
  1800→      get().goHome({ push: false });
  1801→    },
  1802→
  1803→    markSaved: (name) => set({ savedName: name, dirty: false }),
